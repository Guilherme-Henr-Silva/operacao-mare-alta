// lado a lado + quem passou 2x no mesmo terminal
MATCH (k:Container)-[p:PASSOU_POR]->(t:Terminal) WHERE k.classe = 'Perigoso' WITH t, k, count(p) AS vezes RETURN t.nome AS terminal, sum(vezes) AS count_k, count(k) AS count_distinct_k, [x IN collect(CASE WHEN vezes > 1 THEN k.id END) WHERE x IS NOT NULL] AS passaram_2x ORDER BY terminal;
